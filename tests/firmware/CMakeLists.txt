cmake_minimum_required(VERSION 3.24)
project(MAVMOOSFirmwarePrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAVLINK_DIR "" CACHE PATH "Pinned generated MAVLink headers")
if(NOT EXISTS "${MAVLINK_DIR}/ardupilotmega/mavlink.h")
  message(FATAL_ERROR "Supply pinned generated MAVLINK_DIR")
endif()
add_executable(firmware_discovery_tests FirmwareDiscoveryTests.cpp ../../src/libmavmoos_runtime/FirmwareDiscovery.cpp)
target_include_directories(firmware_discovery_tests PRIVATE ../../src/libmavmoos_runtime)
target_include_directories(firmware_discovery_tests SYSTEM PRIVATE "${MAVLINK_DIR}")
target_compile_options(firmware_discovery_tests PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME firmware_discovery COMMAND firmware_discovery_tests)
set_tests_properties(firmware_discovery PROPERTIES TIMEOUT 10)
add_test(NAME firmware_manifest COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/ManifestTests.py")
set_tests_properties(firmware_manifest PROPERTIES TIMEOUT 30)
