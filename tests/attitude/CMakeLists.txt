cmake_minimum_required(VERSION 3.24)
project(MAVMOOSAttitudePrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MAVLINK_DIR)
 message(FATAL_ERROR "Supply MAVLINK_DIR with immutable generated ardupilotmega headers")
endif()
add_library(attitude_provider MODULE "${REPO_ROOT}/src/plugins/AttitudePlugin.cpp")
target_include_directories(attitude_provider PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(attitude_provider SYSTEM PRIVATE "${MAVLINK_DIR}")
target_compile_options(attitude_provider PRIVATE -Wall -Wextra -Wpedantic)
add_executable(attitude_scenario_driver AttitudeScenarioDriver.cpp)
target_include_directories(attitude_scenario_driver PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(attitude_scenario_driver SYSTEM PRIVATE "${MAVLINK_DIR}")
target_link_libraries(attitude_scenario_driver PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(attitude_scenario_driver PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME attitude_provider COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/AttitudeProviderTests.py" --driver $<TARGET_FILE:attitude_scenario_driver> --module $<TARGET_FILE:attitude_provider>)
set_tests_properties(attitude_provider PROPERTIES TIMEOUT 20)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/attitude-targets.json" CONTENT "{\"driver\":\"$<TARGET_FILE:attitude_scenario_driver>\",\"module\":\"$<TARGET_FILE:attitude_provider>\"}\n")
