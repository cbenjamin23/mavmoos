cmake_minimum_required(VERSION 3.24)
project(MAVMOOSExternalObserverMission LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MOOS_IVP_ROOT "$ENV{MOOS_IVP_ROOT}" CACHE PATH "Built stock MOOS-IvP checkout")
find_package(MOOS 10.0 REQUIRED CONFIG PATHS
  "${MOOS_IVP_ROOT}/build/MOOS/MOOSCore" "${MOOS_IVP_ROOT}/build/MOOS" NO_DEFAULT_PATH)
add_executable(external_observer_probe ExternalObserverProbe.cpp)
target_link_libraries(external_observer_probe PRIVATE ${MOOS_LIBRARIES})
target_compile_options(external_observer_probe PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME external_observer_evidence COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/EvidenceTests.py")
set_tests_properties(external_observer_evidence PROPERTIES TIMEOUT 15)
