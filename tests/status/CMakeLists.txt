cmake_minimum_required(VERSION 3.24)
project(MAVMOOSStatusPrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MAVLINK_DIR)
 message(FATAL_ERROR "Supply MAVLINK_DIR with immutable generated ardupilotmega headers")
endif()
add_library(status_provider MODULE "${REPO_ROOT}/src/plugins/SystemStatusPlugin.cpp")
target_include_directories(status_provider PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(status_provider SYSTEM PRIVATE "${MAVLINK_DIR}")
target_compile_options(status_provider PRIVATE -Wall -Wextra -Wpedantic)
add_executable(status_scenario_driver StatusScenarioDriver.cpp)
target_include_directories(status_scenario_driver PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(status_scenario_driver SYSTEM PRIVATE "${MAVLINK_DIR}")
target_link_libraries(status_scenario_driver PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(status_scenario_driver PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME status_provider COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/StatusProviderTests.py" --driver $<TARGET_FILE:status_scenario_driver> --module $<TARGET_FILE:status_provider>)
set_tests_properties(status_provider PROPERTIES TIMEOUT 20)
add_test(NAME status_observation_audit COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/StatusObservationAuditTests.py")
set_tests_properties(status_observation_audit PROPERTIES TIMEOUT 10)
if(MOOS_IVP_ROOT)
 find_package(MOOS 10.0 REQUIRED CONFIG PATHS "${MOOS_IVP_ROOT}/build/MOOS/MOOSCore" "${MOOS_IVP_ROOT}/build/MOOS" NO_DEFAULT_PATH)
 add_executable(status_moos_probe StatusMoosProbe.cpp)
 target_link_libraries(status_moos_probe PRIVATE ${MOOS_LIBRARIES})
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/status-targets.json" CONTENT "{\"driver\":\"$<TARGET_FILE:status_scenario_driver>\",\"module\":\"$<TARGET_FILE:status_provider>\"}\n")
