cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(MAVMOOSConfigurationConsumerTests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  get_filename_component(default_source "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  set(MAVMOOS_SOURCE_ROOT "${default_source}" CACHE PATH "Exact matching host checkout")
  include(FetchContent)
  set(MAVMOOS_MAVLINK_ARCHIVE "" CACHE FILEPATH "Offline pinned generated MAVLink source archive")
  set(mavlink_url "https://codeload.github.com/mavlink/c_library_v2/tar.gz/a2d0ecf011a4aaa7840480e43f4e38d8c9e79d29")
  if(MAVMOOS_MAVLINK_ARCHIVE)
    set(mavlink_url "${MAVMOOS_MAVLINK_ARCHIVE}")
  endif()
  FetchContent_Declare(mavlink URL "${mavlink_url}"
    URL_HASH SHA256=3fc41a3cd46717b1aa226b958aed0829fbe7b7c9793bc8ffdcca76345a9dfde3
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(mavlink)
  add_library(mavlink_headers INTERFACE)
  target_include_directories(mavlink_headers SYSTEM INTERFACE "${mavlink_SOURCE_DIR}")
  option(MAVMOOS_CONFIGURATION_SANITIZERS "Address/undefined instrumentation for matching host and fixtures" OFF)
  if(MAVMOOS_CONFIGURATION_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
  add_subdirectory("${MAVMOOS_SOURCE_ROOT}/src/libmavmoos_runtime" "${CMAKE_BINARY_DIR}/matching-runtime")
  include(CTest)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()

foreach(kind alpha bravo default provider abi9)
  add_library(configuration_${kind} MODULE ConfigurationFixture.cpp)
  set_target_properties(configuration_${kind} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
  target_include_directories(configuration_${kind} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(configuration_${kind} PRIVATE mavmoos_runtime)
  target_compile_definitions(configuration_${kind} PRIVATE CONFIGURATION_FIXTURE_NAME="configuration_${kind}")
  target_compile_options(configuration_${kind} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
target_compile_definitions(configuration_default PRIVATE CONFIGURATION_DEFAULT_ONLY)
target_compile_definitions(configuration_provider PRIVATE CONFIGURATION_PROVIDER)
target_compile_definitions(configuration_abi9 PRIVATE CONFIGURATION_ABI9)
add_executable(plugin_configuration_tests ConfigurationTests.cpp)
target_link_libraries(plugin_configuration_tests PRIVATE mavmoos_runtime ${CMAKE_DL_LIBS})
target_compile_options(plugin_configuration_tests PRIVATE -Wall -Wextra -Wpedantic)
add_executable(plugin_configuration_source_abi_probe SourceAbiProbe.cpp)
target_link_libraries(plugin_configuration_source_abi_probe PRIVATE mavmoos_runtime ${CMAKE_DL_LIBS})
target_compile_options(plugin_configuration_source_abi_probe PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME plugin_configuration COMMAND plugin_configuration_tests
  $<TARGET_FILE:configuration_alpha> $<TARGET_FILE:configuration_bravo>
  $<TARGET_FILE:configuration_default> $<TARGET_FILE:configuration_provider> $<TARGET_FILE:configuration_abi9>)
set_tests_properties(plugin_configuration PROPERTIES TIMEOUT 20)
