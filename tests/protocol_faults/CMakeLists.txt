cmake_minimum_required(VERSION 3.24)
project(MAVMOOSProtocolFaultPrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "Only this focused suite" FORCE)
add_subdirectory("${REPO_ROOT}" "${CMAKE_BINARY_DIR}/mavmoos")
add_executable(protocol_fault_tests ProtocolFaultTests.cpp)
target_link_libraries(protocol_fault_tests PRIVATE mavmoos_runtime)
target_compile_options(protocol_fault_tests PRIVATE -Wall -Wextra -Wpedantic)
add_executable(protocol_native_driver ProtocolNativeDriver.cpp)
target_link_libraries(protocol_native_driver PRIVATE mavmoos_runtime mavmoos_rx)
target_compile_options(protocol_native_driver PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME protocol_faults COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/run_fake.py" --driver $<TARGET_FILE:protocol_fault_tests> --parameters $<TARGET_FILE:mavmoos_parameters> --missions $<TARGET_FILE:mavmoos_missions> --control $<TARGET_FILE:mavmoos_arducopter_control>)
set_tests_properties(protocol_faults PROPERTIES TIMEOUT 20)
add_test(NAME protocol_native_audit COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/test_native_audit.py")
set_tests_properties(protocol_native_audit PROPERTIES TIMEOUT 10)
