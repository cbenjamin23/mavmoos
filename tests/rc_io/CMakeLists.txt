cmake_minimum_required(VERSION 3.24)
project(MAVMOOSRcIoPrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MAVLINK_DIR)
 message(FATAL_ERROR "Supply MAVLINK_DIR with immutable generated ardupilotmega headers")
endif()
add_library(rc_io_provider MODULE "${REPO_ROOT}/src/plugins/RcIoPlugin.cpp")
target_include_directories(rc_io_provider PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(rc_io_provider SYSTEM PRIVATE "${MAVLINK_DIR}")
target_compile_options(rc_io_provider PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rc_io_scenario_driver RcIoScenarioDriver.cpp)
target_include_directories(rc_io_scenario_driver PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(rc_io_scenario_driver SYSTEM PRIVATE "${MAVLINK_DIR}")
target_link_libraries(rc_io_scenario_driver PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(rc_io_scenario_driver PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME rc_io_provider COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/RcIoProviderTests.py" --driver $<TARGET_FILE:rc_io_scenario_driver> --module $<TARGET_FILE:rc_io_provider>)
set_tests_properties(rc_io_provider PROPERTIES TIMEOUT 20)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/rc-io-targets.json" CONTENT "{\"driver\":\"$<TARGET_FILE:rc_io_scenario_driver>\",\"module\":\"$<TARGET_FILE:rc_io_provider>\"}\n")
