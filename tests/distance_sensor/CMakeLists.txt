cmake_minimum_required(VERSION 3.24)
project(MAVMOOSDistancePrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MAVLINK_DIR)
 message(FATAL_ERROR "Supply MAVLINK_DIR with immutable generated ardupilotmega headers")
endif()
add_library(distance_sensor_provider MODULE "${REPO_ROOT}/src/plugins/DistanceSensorPlugin.cpp")
target_include_directories(distance_sensor_provider PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(distance_sensor_provider SYSTEM PRIVATE "${MAVLINK_DIR}")
target_compile_options(distance_sensor_provider PRIVATE -Wall -Wextra -Wpedantic)
add_executable(distance_sensor_scenario_driver DistanceSensorScenarioDriver.cpp)
target_include_directories(distance_sensor_scenario_driver PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(distance_sensor_scenario_driver SYSTEM PRIVATE "${MAVLINK_DIR}")
target_link_libraries(distance_sensor_scenario_driver PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(distance_sensor_scenario_driver PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME distance_sensor_provider COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/DistanceSensorProviderTests.py" --driver $<TARGET_FILE:distance_sensor_scenario_driver> --module $<TARGET_FILE:distance_sensor_provider>)
set_tests_properties(distance_sensor_provider PROPERTIES TIMEOUT 20)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/distance-targets.json" CONTENT "{\"driver\":\"$<TARGET_FILE:distance_sensor_scenario_driver>\",\"module\":\"$<TARGET_FILE:distance_sensor_provider>\"}\n")
