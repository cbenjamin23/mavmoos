cmake_minimum_required(VERSION 3.24)
project(MAVMOOSObserverClockPrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MAVLINK_DIR)
  message(FATAL_ERROR "Supply immutable generated MAVLINK_DIR")
endif()
if(NOT RUNTIME_SOURCE_ROOT)
  set(RUNTIME_SOURCE_ROOT "${REPO_ROOT}")
endif()
set(runtime_dir "${RUNTIME_SOURCE_ROOT}/src/libmavmoos_runtime")
add_library(observer_test_runtime "${runtime_dir}/PluginHost.cpp" "${runtime_dir}/Runtime.cpp"
  "${runtime_dir}/RuntimeMission.cpp" "${runtime_dir}/RuntimeStream.cpp"
  "${runtime_dir}/ObservationRequests.cpp" "${runtime_dir}/ModuleSelection.cpp")
target_include_directories(observer_test_runtime PUBLIC "${runtime_dir}")
target_include_directories(observer_test_runtime SYSTEM PUBLIC "${MAVLINK_DIR}")
target_link_libraries(observer_test_runtime PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(observer_test_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(observer_runtime_clock_tests ObserverRuntimeClockTests.cpp)
target_link_libraries(observer_runtime_clock_tests PRIVATE observer_test_runtime)
target_compile_options(observer_runtime_clock_tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
if(ATTITUDE_MODULE AND STATUS_MODULE AND RC_MODULE)
  add_test(NAME observer_runtime_clock COMMAND observer_runtime_clock_tests "${ATTITUDE_MODULE}" "${STATUS_MODULE}" "${RC_MODULE}")
  set_tests_properties(observer_runtime_clock PROPERTIES TIMEOUT 10)
endif()
