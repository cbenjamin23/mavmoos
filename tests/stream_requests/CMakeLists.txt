cmake_minimum_required(VERSION 3.24)
project(MAVMOOSStreamRequestsPrivate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT MAVLINK_DIR)
  message(FATAL_ERROR "Supply MAVLINK_DIR with immutable generated headers")
endif()
add_executable(stream_request_tests StreamRequestTests.cpp
  "${REPO_ROOT}/src/libmavmoos_runtime/ObservationRequests.cpp")
target_include_directories(stream_request_tests PRIVATE "${REPO_ROOT}/src/libmavmoos_runtime")
target_include_directories(stream_request_tests SYSTEM PRIVATE "${MAVLINK_DIR}")
target_compile_options(stream_request_tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME stream_request_state COMMAND stream_request_tests)
set_tests_properties(stream_request_state PROPERTIES TIMEOUT 10)
